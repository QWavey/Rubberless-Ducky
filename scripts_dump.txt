    function makeSnippets(snippets) {
        return (Array.isArray(snippets) ? snippets : [snippets])
            .map(({ name, code }) =>
                [
                    'snippet ' + name,
                    code
                        .split('\\n')
                        .map(c => '\t' + c)
                        .join('\n'),
                ].join('\n')
            )
            .join('\n')
    }

    function updateExtension(foldRowStart){
        let range = getEditor().getSession().getFoldWidgetRange(foldRowStart);
        // range.start.row = range.start.row;
        range.end.row = range.end.row+1;
        range.end.column = range.end.column+1;
        range.start.column = 0
        let payloadName = trimInput(getEditor().getSession().getLine(foldRowStart));
        editor.selection.setRange(range);
        snippetManager.insertSnippet(editor, snippetManager.snippetMap['hak5duckyscript'].filter(s => s.name === payloadName)[0].content);
        successToast(payloadName + ' Updated!', 'Payload contents modified');
        clearAnnotations('extension');
    }

    function attachBreakpointHandler(){
            getEditor().on("guttermousedown", function(e) {
                /*  */
                let arow =  e.getDocumentPosition().row;
                let anno = getEditor().getSession().getAnnotations().filter(a => a.row === arow);
                if (anno !== [] && anno[0] !== undefined) {
                    let skipAnnoOpen = false;
                    try{
                        if (e.domEvent.target.classList[0] === 'ace_fold-widget'){
                            skipAnnoOpen = true;
                            eLog("Fold Widget - open annotation skipped");
                        }
                    } catch{
                        console.warn("Fold Widget Error");
                    }
                    if(!skipAnnoOpen) {
                        getEditor().gotoLine(e.getDocumentPosition().row+1,0,false);
                        getEditor().execCommand("goToNextError");
                        let payloadName = trimInput(getEditor().getSession().getLine(arow)).replace("EXTENSION","");
                        if (anno[0].text.includes('OUT-OF-DATE')){
                            buttonToast('OUT OF DATE','Update ' + payloadName +' Extension with the latest official version?','Update Extension',function(){updateExtension(arow)},0);
                            return;
                        }
                        if (anno[0].text.includes('MODIFIED')){
                            buttonToast('MODIFIED','Replace ' + payloadName + ' Extension with the latest official version?','Update Extension',function(){updateExtension(arow)},0);
                            return;
                        }
                        if (anno[0].text.includes('UNOFFICIAL')){
                            buttonToast('UNOFFICIAL', payloadName + ' Extension unrecognized - Only use trusted and known code in your payloads.', 'Remove Unofficial Extension?');
                            return;
                        }
                        if (anno[0].text.includes('UP-TO-DATE')&& anno[0].text.includes('OFFICIAL')){
                            successToast("UP TO DATE", payloadName + ' Extension matches official latest version',4500,'toast--top-right');
                        }
                    }
                }
            });
    }

    async function doEditorChange(){
            eLog('ON CHANGE');
            /*  */

            onChangeRunning = true;
            fixHeight();
            if (IDE_CONFIG['autosavesession']) {
                markDirty();
                clearTimeout(sessionSaver);
                sessionSaver = setTimeout(saveSessionToBrowser, 1000);
            }

            downloadDirtyCheck();

            /*  */

            if (IDE_CONFIG['editor_mode'].id === MODE_DUCK.id) {
                if (IDE_CONFIG['validateextensions']) {
                    if (!extensionChecker && !working) {
                        clearTimeout(extensionChecker);
                        extensionChecker = setTimeout(() => {
                            detectExtensions();
                            extensionChecker = null;
                            working = false;
                        }, 1000);
                    } else {
                        eLog("ONCHANGE: already working:" + working + " already scheduled:" + extensionChecker, 'warn')
                    }
                }
            } else {
                clearTimeout(extensionChecker);
                working = false;
            }
            onChangeRunning = false;
    }

            if (!pasteFixer){
                pasteFixer = true;
                eLog("paste detected - forcing newline mode");
                let rr = getEditor().getSession().getSelection().cursor.row+1;
                let rc = getEditor().getSession().getSelection().cursor.column;
                gotoline(0);
                setTimeout(() => {
                    gotoline(rr,rc);
                    setTimeout(() => {
                        detectExtensions();
                    }, 15);
                }, 15);
                editor.setOption("newLineMode", 'unix');
                pasteFixer = null;
            } else {
                eLog("ONPASTE paste fixer already working",'warn');
            }
        });
    }

    function r(min, max) {
        min = Math.ceil(min);
        //console.warn(Math.floor(Math.random() * (max - min) + min));
        max = Math.floor(max);
        return Math.floor(Math.random() * (max - min) + min);
    }

    function sessionToJSON(session) {
        return {
            selection: session.selection.toJSON(),
            value: session.getValue(),
            history: {
                undo: session.$undoManager.$undoStack.map(filterHistory),
                redo: session.$undoManager.$redoStack.map(filterHistory)
            },
            breakpoints: serializeBreakpoints(session.getBreakpoints()),
            scrollTop: session.getScrollTop(),
            scrollLeft: session.getScrollLeft(),
            options: session.getOptions()
        }
    }

    function sessionFromJSON(data) {
        if (!data) {
            eLog('no session to load');
            return;
        }
        var session = ace.require("ace/ace").createEditSession(data.value);
        session.$undoManager.$doc = session;
        setEditorOptions();

        /*  */

        session.$undoManager.$undoStack = data.history.undo;
        session.$undoManager.$redoStack = data.history.redo;
        session.selection.fromJSON(data.selection);
        session.setScrollTop(data.scrollTop);
        session.setScrollLeft(data.scrollLeft);
        return session;
    };

        if (IDE_CONFIG['viewing_mode'] === true){

            eLog("Console Session Saved");// + JSON.stringify(sessionData));

            if (!consoleOnly) {
                var session = getEditor().getSession()
                var sessionData = sessionToJSON(session);
                // safely handles circular references
                JSON.safeStringify = (obj, indent = 2) => {
                    let cache = [];
                    const retVal = JSON.stringify(
                        obj,
                        (key, value) =>
                            typeof value === "object" && value !== null
                                ? cache.includes(value)
                                    ? undefined// Duplicate reference found, discard key
                                    : cache.push(value) && value // Store value in our collection
                                : value,
                        indent
                    );
                    cache = null;
                    return retVal;
                };
                localStorage.PSES = JSON.safeStringify(sessionData);
                eLog("Editor Session Saved");// + JSON.stringify(sessionData));
            }
            markClean();
        } catch(e) {
            console.warn(e+' session save failed');
            flashMessage('Error Saving Session');
        }
    }

    function CONFIG_MIGRATIONS(src) {
        let dest = IDE_CONFIG;
        let upgradeModeModel = toMode(src['editor_mode'].id);
        for (const [key, value] of Object.entries(dest)) {
            if (key === 'VERSION'){
                dest[key] = ENCODER_VERSION;
                continue;
            }
            let potentialValue = src[key];
            if (potentialValue !== null && potentialValue !== undefined && potentialValue !== "" && potentialValue !== {}){
                if (value !== null && value !== undefined && value !== "" && value !== {}){
                    if (dest[key] !== src[key]){
                        dest[key] = src[key];
                        console.info('Migrated Existing config: '+key+ ":"+dest[key]);
                    }
                }
            }
        }
        dest['editor_mode'] = upgradeModeModel;
        dest['editor_mode'].label = src['editor_mode'].label;
        dest['editor_mode'].langdef = src['editor_mode'].langdef;
        return dest;
    }

    function detectExtensions(){
        try {
            eLog("Detect Extensions");
            var extensionsDetected = 0;
            extensions = [];
            if (!working) {
                working = true;
                var folds = getEditor().getSession().foldWidgets;
                if (!folds){
                    //console.warn("detectExtensions Cancelled");
                    working = false;
                    return;
                }
                let foldStarts = [];
                for (let i = 0; i < folds.length; i++){
                    if (folds[i] === 'start'){
                        foldStarts.push(i);
                    }
                }
                for (let i = 0; i < foldStarts.length; i++){
                    let foldRowStart = foldStarts[i];
                    let range = getEditor().getSession().getFoldWidgetRange(foldRowStart);
                    let rowData = trimInput(getEditor().getSession().getLine(foldRowStart));
                    if (/^EXTENSION\ .*/.test(rowData)){
                        extensionsDetected++;
                        //console.log("check extension: "+rowData);
                        try {
                            let data = determineExtensionState(foldRowStart+2,range,rowData);
                            //console.log(data);
                            let tags = data['tags'].join(' ');
                            //console.warn(tags);
                            //console.warn(data);
                            let level = 'info';
                            if (/MODIFIED/.test(tags)){
                                level = 'warning';
                            }
                            if (/OUT-OF-DATE/.test(tags)){
                                level = 'error';
                            }
                            extensions.push({'data': data, 'frs': foldRowStart, 'level': level});
                        }catch {

                        }
                    }
                }
                clearAnnotations('extension',"detectExtensions");
                for (let h = 0; h < extensions.length; h++){
                    let extension = extensions[h];
                    addAnnotation('extension',extension['frs'],extension['data']['tags'],extension['level']);
                }
                eLog('Extensions detected '+ extensionsDetected);
                working = false;
            }
        } catch {
            //console.error("ERROR SYNCING");
        } finally {
            working = false;
        }
    }

    function determineExtensionState(foldRowStart,range,rowData) {
        var matchesOfficialExtensionName = false;
        var tags = [];
        var payloadExtensionName = '';
        var officialExtensionName = '';
        var payloadExtensionVersion = '';
        var officialExtensionVersion = '';
        var officialCodeBodyLen = 0;
        var payloadCodeBody = '';
        var officialCodeBody = '';
        var payloadCodeBodyLen = 0;
        for (let k = 0; k < _snippetsMap.length; k++) {
            officialExtensionName  = _snippetsMap[k]["extension_name"];
            if (!officialExtensionName){
                continue;
            }
            payloadExtensionName = rowData;

            officialCodeBody = _snippetsMap[k]['code'].split("\\n");
            officialExtensionVersion = trimInput(_snippetsMap[k]['code'].split("\\n").splice(1,1).join('').replace('REM VERSION ', ''));
            officialCodeBody = officialCodeBody.splice(2,officialCodeBody.length-4);
            officialCodeBodyLen = officialCodeBody.length;
            officialCodeBody = officialCodeBody.join('\n').replaceAll('\\$','$');

            payloadCodeBody = getEditor().getSession().getLines(foldRowStart,range['end']['row']);
            payloadCodeBodyLen = payloadCodeBody.length;
            payloadCodeBody = payloadCodeBody.join('\n');
            payloadExtensionVersion = trimInput(getEditor().getSession().getLines(foldRowStart-1,foldRowStart-1).join('').replace('REM VERSION\ ','').replace(/\-.*$/,''));
            if (officialExtensionName === payloadExtensionName) {
                // Name Matches
                tags.push('OFFICIAL');
                matchesOfficialExtensionName = true;
                if (officialExtensionVersion === payloadExtensionVersion) {
                    // Version Matches
                    tags.push('UP-TO-DATE');
                    if (payloadCodeBody !== officialCodeBody){
                        // Content DOESNT Match
                        //console.error(payloadCodeBody +" \nvs\n "+ officialCodeBody);
                        tags.unshift('MODIFIED');

                    }
                } else {
                    // Version DOES NOT Match
                    //console.error(officialExtensionVersion + "vs" + payloadExtensionVersion);
                    tags.push('OUT-OF-DATE');
                }
                break;


            }
        }
        if (!matchesOfficialExtensionName) {
            // Unofficial
            tags.push('UNOFFICIAL');
        }
        return {'tags':tags, 'payloadExtensionName': payloadExtensionName,'payloadExtensionVersion': payloadExtensionVersion, 'officialExtensionName': officialExtensionVersion, 'officialExtensionVersion': officialExtensionVersion };
    }

    var clearLock = false;
    function clearAnnotations(t,caller="") {
        if (clearLock) {
            eLog("Clear Annotations LOCKED");
            return;
        }
        clearLock = true;
        eLog("Clear annotations: " + t + " from " + caller);
        let ret = [];
        let markers = getEditor().getSession().getAnnotations();
        for (let i = 0; i < markers.length; i++) {
            let marker = markers[i];
            if (marker['a_type'] !== t) {
                ret.push(marker);
            }
        }
        editor.getSession().setAnnotations(ret);

        let retm = [];
        let actualmarkers = getEditor().getSession().getMarkers(true);
        for (let i = 0; i < actualmarkers.length; i++) {
            let actualmarker = actualmarkers[i];
            if (actualmarker) {
                if (actualmarker['a_type'] !== t) {
                    retm.push(actualmarker);
                } else {
                    editor.getSession().removeMarker(actualmarker.id);
                }
            }
        }

        editor.getSession().$frontMarkers = retm;
        editor.renderer.updateFrontMarkers();
        editor.renderer.updateBackMarkers();

        /*  */
        clearLock = false;
    }

    function addAnnotation(t,row,data,level,appendData=false){
        var annotations = getEditor().getSession().getAnnotations();
        let found = false;
        for (let i = 0; i < annotations.length; i++){
            if (annotations[i]['row'] === row){
                appendData ? annotations[i]['text']+="\n" + data : annotations[i]['text']=data;
                if(annotations[i]['type'] === 'error'){
                    annotations[i]['type'] = 'error';
                } else if (annotations[i]['type'] === 'warning') {
                  if (level === "error") {
                      annotations[i]['type'] = level;
                  }
                }
                annotations[i]['type'] = level;
                found = true;
                break;
            }
        }
        if (!found){
            annotations.push({'a_type':t,'type': level, "row": row, "column": 0,  "text": data});
            //if (t !== "compilerDEBUG") {
            //    eLog('add Annotation: LINE '+row + " " +JSON.stringify(data));
            //}
        }
        getEditor().getSession().setAnnotations(annotations);

        /*  */
    }

    function gatherDefines(lines) {
        const caller = "DEFINE PREPROCESSOR";
        for (let i = 0; i < lines.length; i++ ) {
            let line = lines[i];
            if (trimInput(line).match(rem_block_regex)){
                INSIDE_REM_BLOCK = true;
                continue;
            } else if (trimInput(line).match(end_rem_block_regex)){
                INSIDE_REM_BLOCK = false;
                continue;
            }
            if (trimInput(line).match(define_regex)){
                let words = trimInput(line).split(" ");
                let label = words[1];
                let replacement = trimInput(line).split("DEFINE " + label + " ")[1];
                dLog(caller, "DEFINE label: " + label,false);
                dLog(caller, "DEFINE replacement: "  + replacement,false);
                DS3();
                if (preprocessorLabels.includes(label)){
                    throwError("Duplicate DEFINE Label: "+label,i);
                } else {
                    preprocessorLabels.push(label);
                    preprocessorReplacements.push(replacement);
                    //     line = "REM "+line; removed
                }
            }
            lines[i] = line;
        }
        return lines;
    }

    function searchLabel(label,match){
        for (let n = 0; n < preprocessorLabels.length; n++) {
            let l = preprocessorLabels[n];
            if (label === l) {
                return preprocessorReplacements[n] === match;
            }
        }
        return match === "FALSE";
    }

    function parseIfDefs(lines){
        let ifdef_stack = [];
        for (let i = 0; i < lines.length; i++ ) {
            let line = lines[i];
            if (line === ""){
                continue;
            }
            let label = trimInput(line).split(" ")[1];
            let found = false;
            let skip = false;

            switch (true) {
                case ifdef_regex.test(line):
                    found = searchLabel(label,"TRUE");
                    ifdef_stack.push([trimInput(line),found]);
                    break;
                case ifnotdef_regex.test(line):
                    found = !searchLabel(label,"FALSE");
                    ifdef_stack.push([trimInput(line),found]);
                    break;
                case elsedef_regex.test(line):
                    //NEGATE
                    if (ifdef_stack.length === 0) {
                        throwError("No IF_DEFINED_TRUE or IF_NOT_DEFINED_TRUE to negate",i);
                        return;
                    }
                    let mode = ifdef_stack[ifdef_stack.length-1][1];
                    let m = ifdef_stack[ifdef_stack.length-1][0]
                    let v = !ifdef_stack[ifdef_stack.length-1][1]
                    ifdef_stack.pop();
                    ifdef_stack.push([m,!mode]);
                    break;
                case endifdef_regex.test(line):
                    //POP
                    ifdef_stack.pop();
                    break;
                default:
                    break;
            }


            if(ifdef_stack.length > 0){
                let insideEnabledCode = true;
                let currentval;
                let mode;
                let l;
                for (l = 0; l < ifdef_stack.length; l++){
                   currentval = ifdef_stack[l][1];
                   mode = ifdef_stack[l][0];
                   switch (true) {
                       case ifdef_regex.test(mode):
                           if (!currentval){
                               insideEnabledCode = false;
                           }
                           break;
                       case ifnotdef_regex.test(mode):
                           if (currentval){
                               insideEnabledCode = false;
                           }
                           break;
                   }
                   if (!insideEnabledCode){
                       break;
                   }
                }
                if (!insideEnabledCode){
                    //throwWarn("This line will not be included in the payload",i);
                    if (line !== mode) {
                        addAnnotation('compilerDEBUG', i, "This line wont be included because " + mode + " does not evaluate to TRUE\n", 'info', true);
                    }
                    lines[i] = "PREPROCESSOR_DISABLED " + line;
                }
            }
        }
        return lines;
    }

    function handleIf(line,chain=false) {
        DS3();
        let caller = "handleIf()";
        dLog(caller, line);
        nextAvailableBlockAddress++;
        currentBlock = nextAvailableBlockAddress;
        if (!chain){
            awaitingFutureAddrAssignment.push([]);
            dLog(caller,"[New If Chain]: " + awaitingFutureAddrAssignment)
        }
        let stack = [];
        let outputStack = [];
        let lpCount = 0;
        let rpCount = 0;
        let lexemes = line.split(" ");
        dLog('[*] Parsing', '\t' + lexemes.join(' '));
        /*  */
        for (let i = 0; i < lexemes.length; i++) {
            let word = lexemes[i];
            switch (true) {
                case /^\s*IF\s*$/.test(word):
                    break;
                case containsFunctionCall(word):
                    nextRegister();
                    const returnReg = currentRegister();
                    stack.push(returnReg);
                    handleFunctionCall(word);
                    appendHexStringArray(handleAssignment(returnReg + " = $_f_ret"));
                    freeEvalRegisters--;
                    break;
                case isReservedConstant(word):
                    stack.push(word);
                    break;
                case isReservedVar(word):
                    stack.push(word);
                    break;
                case isVar(word):
                    stack.push(word);
                    break;
                case 'THEN' === word || '{' === word:
                    break;
                case '(' === word:
                    lpCount++;
                    break;
                case ')' === word:
                    rpCount++;
                    let temp = [];
                    nextRegister();
                    if (stack.length > 1) {
                        let var2 = stack.splice(stack.length - 1, 1)[0];
                        let op = stack.splice(stack.length - 1, 1)[0];
                        let var1 = stack.splice(stack.length - 1, 1, currentRegister())[0];
                        temp.push("=");
                        temp.push(currentRegister());
                        temp.push(var1);
                        temp.push(var2);
                        temp.push(op);
                    } else if (stack.length === 0) {
                        throwError("Empty Expresion");
                    } else {
                        let var1 = stack.splice(stack.length - 1, 1, currentRegister())[0];
                        temp.push("=");
                        temp.push(currentRegister());
                        temp.push(var1);
                        temp.push("0000");
                    }
                    freeEvalRegisters--;
                    outputStack.push(temp);
                    break;
                case '^' === word:
                    stack.push(word);
                    break;
                case '<' === word:
                    stack.push(word);
                    break;
                case '>' === word:
                    stack.push(word);
                    break;
                case '<=' === word:
                    stack.push(word);
                    break;
                case '>=' === word:
                    stack.push(word);
                    break;
                case '>>' === word:
                    stack.push(word);
                    break;
                case '<<' === word:
                    stack.push(word);
                    break;
                case '==' === word:
                    stack.push(word);
                    break;
                case '!=' === word:
                    stack.push(word);
                    break;
                case '-' === word:
                    stack.push(word);
                    break;
                case '+' === word:
                    stack.push(word);
                    break;
                case '*' === word:
                    stack.push(word);
                    break;
                case '%' === word:
                    stack.push(word);
                    break;
                case '/' === word:
                    stack.push(word);
                    break;
                case '&' === word:
                    stack.push(word);
                    break;
                case '|' === word:
                    stack.push(word);
                    break;
                case ('&&' === word || 'AND' === word):
                    stack.push(word);
                    break;
                case ('||' === word || 'OR' === word):
                    stack.push(word);
                    break;
                case !isNaN(word):
                    stack.push(word);
                    break;
                case isHex(word):
                    stack.push(word);
                    break;
                default:
                    throwError("Unexpected Symbol: " + word);
                    break;
            }
        }
        if (lpCount !== rpCount) {
            throwError(" Mismatched ( ): " + lpCount + "'(' vs " + rpCount + " ')'");
        }
        if (stack.length === 1) {
            outputStack.push(['IF', stack[0]]);
        } else {
            outputStack.push(['IF', EVAL_VAR + (totalEvalRegisters - 1)]);
        }

        /* Encode */
        let encoded_line = []; // one if line is expanded into individual statements and evaulated
        for (let i = 0; i < outputStack.length; i++) {
            let encoding = [];
            for (let k = 0; k < outputStack[i].length; k++) {
                let word = outputStack[i][k];
                switch (true) {
                    case ('0000' === word):
                        encoding.push('0000');
                        break;
                    case /^\s*IF\s*$/.test(word):
                        encoding.push(hexEncode(0xEFEF));
                        break;
                    case isReservedVar(word):
                        encoding.push(hexEncode(reservedVariables[word]));
                        break;
                    case isReservedConstant(word):
                        encoding.push(hexEncode(reservedConstants[word]));
                        break;
                    case isOperator(word):
                        encoding.push(hexEncode(operatorMap[word]));
                        break;
                    case isVar(word):
                        if (varExists(word)) {
                            encoding.push(decToHex(getVarAddress(word)));
                        } else {
                            if (/\$__r*/.test(word)) { // if this is a programatically generated temporary variable / register
                                encoding.push(decToHex(allocateVar(word)));
                                assignValue(word, '0000');
                            } else {
                                throwError("Unknown Variable: " + word);
                            }
                        }
                        break;
                    case word === "=":
                        encoding.push(hexEncode(0x01E8));
                        break;
                    case isHex(word):
                        if (varExists(word)) {
                            encoding.push(decToHex(getVarAddress(word)));
                        } else {
                            encoding.push(decToHex(allocateVar(word)));
                            assignValue(word, swapHex(formatHex(word)));
                            // assignValue(word, decToHex(word));
                        }
                        break;
                    case isNumeric(word):
                        if (varExists(word)) {
                            encoding.push(decToHex(getVarAddress(word)));
                        } else {
                            encoding.push(decToHex(allocateVar(word)));

                            assignValue(word, hexEncode(word));
                            // assignValue(word, decToHex(word));
                        }
                        break;
                }
            }
            encoded_line.push(encoding);
        }

        /* Debugging */
        dLog(caller, 'DISASSEMBLY RESULT:');
        for (let i = 0; i < outputStack.length; i++) {
            let gl = outputStack[i].join(' ');
            dLog('[+] Generated line', '\t' + gl);
            ///*  */
        }
        dLog(caller, 'ENCODING RESULT:');
        let el = "";
        encoded_line[encoded_line.length - 1].push(decToHex(currentBlock)); // add block id
        for (let i = 0; i < encoded_line.length; i++) {
            let pr = "";
            for (let k = 0; k < encoded_line[i].length; k++) {
                //el += decToHex(encoded_line[i][k]);// + " ";
                el += encoded_line[i][k];// + " ";
                pr += encoded_line[i][k] + " ";
            }
            dLog('[+] Encoded Line', '\t' + pr);
            ///*  */
        }
        REGISTERS_TO_STRING();

        //freeEvalRegisters = totalEvalRegisters;
        blockStack.push(currentBlock);
        return el.match(/.{1,2}/g);
    }

    function handleLabel(line) {
        DS3();
        let caller = "handleLabel()";
        dLog(caller, line);
        let label = line.replace('LABEL ', '');
        if (labelExists(label)) {
            throwError("Duplicate Definition: " + label);
            return;
        }
        dLog(caller, "[LABEL] Created: "+ label + " @" + oBB.length);
        labelMap[label] = oBB.length;
        // fill in gotos waiting for this label
        if (Object.keys(gotoAwaitingFutureLabel).includes(label)) {
            for (let i = 0; i < gotoAwaitingFutureLabel[label].length; i++) {
                let addr_to_fill_in = gotoAwaitingFutureLabel[label][i];
                dLog(caller, "Filling in GOTO now that label exists:" + label + " @ " + addr_to_fill_in);
                let decAddr = (oBB.length/2);
                let hexAddr = decToHex(decAddr);
                let p1 = hexAddr.substring(0, 2);
                let p2 = hexAddr.substring(2, 4);
                oBB[addr_to_fill_in] = p1;
                oBB[addr_to_fill_in+1] = p2;
            }
        } else if (Object.keys(gotoAwaitingFutureLabel).includes("function_" + label)) {
            for (let i = 0; i < gotoAwaitingFutureLabel["function_" + label].length; i++) {
                let addr_to_fill_in = gotoAwaitingFutureLabel["function_" + label][i];
                dLog(caller, "Filling in GOTO now that label exists:" + label + " @ " + addr_to_fill_in);
                let decAddr = (oBB.length/2);
                let hexAddr = decToHex(decAddr);
                let p1 = hexAddr.substring(0, 2);
                let p2 = hexAddr.substring(2, 4);

    function handleAssignment(line) {
        DS3();
        let caller = "handleAssignment()";
        let encoded_line = [];
        dLog(caller, line);
        let stack = [];
        let outputStack = [];
        let lpCount = 0;
        let rpCount = 0;
        let lexemes = line.split(" ");
        let isDeclaration = false;
        let hasOperation = false;
        let delcaredLiterally = false;
        dLog('[*] Parsing', '\t' + lexemes.join(' '));
        /* Reduction */
        for (let i = 0; i < lexemes.length; i++) {
            let word = lexemes[i];
            switch (true) {
                case /VAR/.test(word):
                    isDeclaration = true;
                    break;
                case containsFunctionCall(word):
                    nextRegister();
                    const returnReg = currentRegister();
                    stack.push(returnReg);
                    handleFunctionCall(word);
                    appendHexStringArray(handleAssignment(returnReg + " = $_f_ret"));
                    freeEvalRegisters--;
                    break;
                case '=' === word:
                    stack.unshift(word);
                    break;
                case isReservedConstant(word):
                    stack.push(word);
                    break;
                case isReservedVar(word):
                    stack.push(word);
                    break;
                case isVar(word):
                    stack.push(word);
                    break;
                case '(' === word:
                    lpCount++;
                    break;
                case ')' === word:
                    rpCount++;
                    let temp = [];
                    if (stack.length > 3) {
                        nextRegister();
                        let var2 = stack.splice(stack.length - 1, 1)[0];
                        let op = stack.splice(stack.length - 1, 1)[0];
                        let var1 = stack.splice(stack.length - 1, 1, currentRegister())[0];
                        temp.push("=");
                        temp.push(currentRegister());
                        temp.push(var1);
                        temp.push(var2);
                        temp.push(op);
                    } else if (stack.length === 0) {
                        throwError("Empty expression");
                    } else {
                        let var1 = stack.splice(stack.length - 1, 1, currentRegister())[0];
                        temp.push("=");
                        temp.push(currentRegister());
                        temp.push(var1);
                    }
                    freeEvalRegisters--;
                    outputStack.push(temp);
                    break;
                case '^' === word:
                    stack.push(word);
                    break;
                case '<' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '>' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '<=' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '>=' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '>>' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '<<' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '==' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '!=' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '-' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '+' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '*' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '%' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '/' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '&' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case '|' === word:
                    hasOperation = true;
                    stack.push(word);
                    break;
                case ('&&' === word || 'AND' === word):
                    hasOperation = true;
                    stack.push(word);
                    break;
                case ('||' === word || 'OR' === word):
                    hasOperation = true;
                    stack.push(word);
                    break;
                case !isNaN(word):
                    stack.push(word);
                    break;
                case isHex(word):
                    stack.push(word);
                    break;
                default:
                    throwError("Unexpected Symbol: " + word);
                    break;
            }
        }

        if (lpCount !== rpCount) {
            throwError("Mismatched ( ): " + lpCount + "'(' vs " + rpCount + " ')'" );
        }
        outputStack.push(stack);

        /* Encode */
        for (let i = 0; i < outputStack.length; i++) {
            let encoding = [];
            for (let k = 0; k < outputStack[i].length; k++) {
                let word = outputStack[i][k];
                switch (true) {
                    //  case word == 0x0000:
                    //    encoding.push(0x0000)
                    //break;
                    case isReservedVar(word):
                        encoding.push(hexEncode(reservedVariables[word]));
                        break;
                    case isReservedConstant(word):
                        encoding.push(hexEncode(reservedConstants[word]));
                        break;
                    case isOperator(word):
                        encoding.push(hexEncode(operatorMap[word]));
                        break;
                    case isVar(word):
                        if (varExists(word)) {
                            encoding.push(decToHex(getVarAddress(word)));
                        } else {
                            encoding.push(decToHex(allocateVar(word)));
                            assignValue(word, '0000');
                        }
                        break;
                    case word === "=":
                        encoding.push(hexEncode(0x01E8));
                        break;
                    case isHex(word):
                        if (varExists(word)) {
                            encoding.push(decToHex(getVarAddress(word)));
                        } else {
                            encoding.push(decToHex(allocateVar(word)));
                            assignValue(word, swapHex(formatHex(word)));
                            // assignValue(word, decToHex(word));
                        }
                        break;
                    case isNumeric(word):
                        if (varExists(word)) {
                            encoding.push(decToHex(getVarAddress(word)));
                        } else {
                            encoding.push(decToHex(allocateVar(word)));
                            assignValue(word, hexEncode(word));
                            //assignValue(word, decToHex(word));
                        }
                        break;
                }
            }
            encoded_line.push(encoding);
        }

        let lastLine = outputStack[outputStack.length - 1];
        let lastToken = lastLine[lastLine.length - 1];
        if (!isOperator(lastToken)) {
            encoded_line.push(['0000']);
        }

        /* Debugging */
        dLog(caller, 'DISASSEMBLY RESULT:');
        //dLog(caller,JSON.stringify(outputStack, undefined, 4));
        for (let i = 0; i < outputStack.length; i++) {
            let gl;
            if (Array.isArray(outputStack[i])) {
                gl = outputStack[i].join(' ');
                dLog('[+] Generated line', '\t' + gl);
            } else {
                gl = outputStack[i];
                dLog('[+] Generated line', '\t' + gl);
            }
            ///*  */
        }
        dLog(caller, 'ENCODING RESULT:');
        let el = "";
        for (let i = 0; i < encoded_line.length; i++) {
            let pr = "";
            for (let k = 0; k < encoded_line[i].length; k++) {
                //el += decToHex(encoded_line[i][k]);// + " ";
                //pr += decToHex(encoded_line[i][k]) + " ";
                el += encoded_line[i][k];// + " ";
                pr += encoded_line[i][k] + " ";
            }
            dLog('[+] Encoded Line', '\t' + pr);
            ///*  */
        }
        REGISTERS_TO_STRING();

        //freeEvalRegisters = totalEvalRegisters;
        return el.match(/.{1,2}/g);
    }

    async function doDuckyEncode() {
        if (COMPILER_RUNNING){
            console.error("already compiling");
            return;
        }

        COMPILER_RUNNING = true;
        reset_encoder_state();
        log('');
        log('[*] Compiling '+IDE_CONFIG['current_file_name']+"...",true);
        eLog('[*] Compiling '+IDE_CONFIG['current_file_name']+"...");
        dLog('[*] Compiling '+IDE_CONFIG['current_file_name']+"...");

        duckyscript = getEditor().getValue();
        debugLogs = [];
        oBB = [];
        ds = duckyscript;
        if ((!duckyscript || duckyscript === '')){
            throwError('Payload is empty');
        }
        try {
            const startTime = window.performance.now();
            let sanitized = sanitizeDS(duckyscript);
            let lines = splitDS(sanitized).map((l) => {
                if (l === " ") {
                    return "SPACE";
                } else {
                    return l;
                }
            });
            lines = preprocess(lines);
            dLog("ENCODING PAYLOAD CONTENT", "\n____________________\n\n", false);
            for (let z = 0; z < lines.length; z++){
                dLog("",z + " " + lines[z],false);
            }
            dLog("", "Begin Encoding Payload", false);
            previousPayloadSize = 0;
            for (currentInstructionIndex; currentInstructionIndex < lines.length; currentInstructionIndex++) {
                if (STOP_COMPILING) {
                    throwError("Compiler exited before reaching end of payload");
                    if (!ONLYLINT){
                        flashMessage("DuckyScript compiler exited early");
                        gotoline(currentInstructionIndex);
                    }
                    if (IDE_CONFIG['showremainingspace']){
                        document.getElementById('payloadpercent').innerText = 'Too many errors...';
                    }
                    throw new Error('too many errors');
                }
                currentLine = lines[currentInstructionIndex];
                dLog(null, "", false);
                dLog(null, "---------------------Begin Encoding Line---------------------", false);
                delegateDSCommand(currentLine);
                let delayToAdd = doDelayCheck();
                if (delayToAdd !== null) {
                    appendHexStringArray(delayToAdd)
                }
                dLog(null, "---------------------Finished Encoding Line---------------------", false);
                dLog(null, "", false);
                freeEvalRegisters = totalEvalRegisters;
            }
            postProcessingLint();
            let added_bytes = 0;
            if (requiresLangPack){
                dLog(null, "-----------------PACK LANGUAGE---------------------", false);
                generateLangPack();
            }
            if (varValues.length > 1) {
                dLog(null, "-----------------PACK REGISTERS---------------------", false);
                REGISTERS_TO_STRING();
                added_bytes += 2;
                oBB.unshift('E8E8');
                for (let i = varValues.length - 1; i > 0; i--) {
                    oBB.unshift((varValues[i]));
                    added_bytes++;
                }
                oBB.unshift('E8E8');
                addToStat("Allocated Registers", added_bytes);
            }
            dLog(null, "-----------------SHIFT/ENCODE ADDRESSES---------------------", false);
            let ui16 = oBB.join('').match(/.{1,4}/g);
            if (ui16 === null){
                ui16=[];
                return;
            }
            for (let i = 0; i < ui16.length; i++) {
                if (ui16[i] === "F8F8" || ui16[i] === "F7F7") {

                    let current_addr = ui16[i];

                }
            }
            oBB = ui16;

    function delegateDSCommand(line) {
        let caller = 'delegateLine()';
        let fCall = null;
        if (!line || line === '') {
            return;
        }
        if(INSIDE_REM_BLOCK){
            if (line.match(end_rem_block_regex)){
                handleEndREMBlock(line);
            }
            ignore();
            return;
        }
        if(INSIDE_STRING_BLOCK){
            let matched = line.match(end_string_regex);
            if (matched){
                fCall = handleEndStringBlock;
            } else {
                if (IDE_CONFIG["stringblockheredoc"]){
                    addAnnotation('compilerDEBUG',currentInstructionIndex,"(Compiler Setting) HEREDOC MODE: All leading whitespace preserved\n STRING blocks strip newlines",'info',true);
                } else {
                    if (!PRESERVE_LEADING_SPACE) {
                        line = line.trimStart();
                        addAnnotation('compilerDEBUG',currentInstructionIndex,"STRING blocks strip leading whitespace \nSTRING blocks strip newlines\n",'info',true);
                    } else if (!IDE_CONFIG["stringblockheredoc"]){
                        for (let i = STRING_BLOCK_INDENTATION_LEVEL; i > 0; i--) {
                            line = line.replace(/^ {4}/, '');
                        }
                    }
                }
                line = "STRING " + line;
                fCall = handleString;
            }
        } else if (INSIDE_STRINGLN_BLOCK){
            let matched = line.match(end_string_regex);
            if (matched){
                fCall = handleEndStringBlock;
            } else {
                if (!IDE_CONFIG["stringblockheredoc"]) {
                    for (let i = STRING_BLOCK_INDENTATION_LEVEL; i > 0; i--) {
                        line = line.replace(/^ {4}/, '');
                    }
                } else {
                    addAnnotation('compilerDEBUG',currentInstructionIndex,"(Compiler Setting) HEREDOC MODE: All leading whitespace preserved\n",'info',true);
                }
                line = "STRINGLN " + line;
                fCall = handleStringln;
            }
        } else {
            for (let i = 0; i < dsSyntaxMap.length; i++) {
                let matched = line.match(dsSyntaxMap[i]);
                if (matched) {
                    let msg = "[+] SCANNER FOUND for line:\n\t" +line +"\n\t" + dsSyntaxMap[i] + "\n\t" +dsHandlerMap[i].name.toString() +'()';
                    dLog(caller,msg);
                    fCall = dsHandlerMap[i];
                    break;
                }
            }
        }

        if (fCall === null) {
            if (line in BUILTINSMap) {
                /*  */

                appendHexStringArray([hexEncode(BUILTINSMap[line])]);
                return;
            } else {
                /*  */
                if (lookupAndAppend(line) === null) {
                    throwError("Unrecognized syntax - Key not found in language file: " + line);
                }
            }
            return;
        } else {
            /*  */
            dLog(caller, 'Delegating syntax for ' + line);
            let tmpByteTab = fCall.call(this, line); // delegate function call
            if (tmpByteTab) {
                appendHexStringArray(tmpByteTab);
            }
        }

        let math = (((oBB.length + ((varValues.length-1) * 2)) / 16384) * 100);
        dLog("", '-------------------------------------------------------------', false);
        dLog("", 'Payload ' + math + '% full -- ' + (oBB.length + ((varValues.length - 1) * 2)) + " of 16384 bytes used", false);
        dLog("", '-------------------------------------------------------------', false);
        if (math >= 100) {
            throwError("Payload too large");
            STOP_COMPILING = true;
            throw new Error("Payload too large");
        }
    }

    function defaultingHexWarning(k) {
        let caller = "fallbackHexWarning()";
        dLog(caller, "WARN - defaulting 0x00 for char: " + k);
        if ( k === "‘" || k === "’") {
            throwError(k + " not defined in " + IDE_CONFIG['ducky_lang_name'] + "\n\t" + "Did you mean straight single quote: ' ?");
        } else if ( k === "“" || k === "”") {

        } else {

    }

    function lookupAndAppend(command,recursive=false,dontsplit=false) {
        let caller = "lookupAndAppend()";
        if (recursive){
            caller="RECURSIVE-"+caller;
        }

        if (!dontsplit) {
            let splitcommand = command.split(" ");
            let buffer = [];
            let modBuffer = [];
            if (splitcommand.length > 1) {
                dLog(caller, "Split: " + splitcommand.join(","));
                for (let i = 0; i < splitcommand.length; i++) {
                    if (splitcommand[i] === "-") {
                        dLog(caller, "SKIP: " + splitcommand[i]);
                        continue;
                    }
                    let keycodes = getBytesForKey(splitcommand[i], true)
                    if (keycodes == null) {
                        return lookupAndAppend(command, true, true);
                    }
                    if (keycodes[1] !== "00") {
                        if (keycodes.length > 1 && keycodes[0] !== '00') {
                            dLog(caller, "+KEY+MODIFIER: " + splitcommand[i]);
                            let m = Object.keys(IDE_CONFIG['ducky_lang']).find(key => IDE_CONFIG['ducky_lang'][key] === keycodes[1] + ",00,00");
                            /*  */
                                throwError('STRICT COMBOS - PREVENTING IMPLICIT MODIFIER COMBINATION');
                                throwError('IMPLICIT MODIFIER COMBINATION: '+splitcommand[i]+ " already contains " + m);
                                return;
                            /*  */
                            buffer.push([keycodes[0]]);
                            modBuffer.push([keycodes[1]]);
                            continue;
                        }
                        dLog(caller, "+MODIFIER: " + splitcommand[i]);
                        modBuffer.push(lookupAndAppend(splitcommand[i], true));
                    } else {
                        if (keycodes.length > 1 && keycodes[1] !== '00') {
                            console.error('compiler error');
                            /*  */
                        }
                        dLog(caller, "+KEY: " + splitcommand[i]);
                        buffer.push(lookupAndAppend(splitcommand[i], true));
                    }
                }

                let keySum = 0;
                for (let i = buffer.length - 1; i >= 0; i--) {
                    dLog(caller, "COMBINE KEY: " + buffer[i]);
                    keySum += Number("0x" + buffer[i]);
                    dLog(caller, "KEY TOTAL: " + keySum);
                }
                let out = [];
                if (keySum < 16) {
                    appendHexStringArray(['0' + keySum.toString(16)]);
                    out.push('0' + keySum.toString(16));
                } else {
                    appendHexStringArray([keySum.toString(16)]);
                    out.push(keySum.toString(16));
                }

                let modSum = 0;
                for (let i = modBuffer.length - 1; i >= 0; i--) {
                    dLog(caller, "COMBINE MODIFIER: " + modBuffer[i]);
                    modSum += Number("0x" + modBuffer[i]);
                    dLog(caller, "MOD TOTAL: " + modSum);
                }
                if (modSum < 16) {
                    appendHexStringArray(['0' + modSum.toString(16)]);
                    out.push('0' + modSum.toString(16));
                } else {
                    appendHexStringArray([modSum.toString(16)]);
                    out.push(modSum.toString(16));
                }


                /*  */
                return true;

            } else if (splitcommand.length > 0) {
                dLog(caller, "SINGLE KEY/MOD: " + splitcommand[0]);
                command = splitcommand[0];
            }
        }

        dLog(caller, "Key lookup on " + command);
        let keyCodes = getBytesForKey(command);

        if (!(keyCodes !== null && typeof keyCodes !== 'undefined')) {
            if (command === command.charAt(0)) {
                dLog(caller,command+ " first Char not found; defaulting to 0x00 ");
                addAnnotation('compilererror',currentInstructionIndex,"Syntax "+command+" not found, no instructions encoded",'error',true);
                defaultingHexWarning(command);
                return false;
            }
            if(IDE_CONFIG['allowimplicitstring']) {
                dLog(caller, command + " word not found, trying first char " + command.charAt(0));
                throwWarn("Deprecated Syntax will only the first character: " + command.charAt(0));
            } else {
                //throwError("(Deprecated Syntax)");
                return null;
            }
            return lookupAndAppend(command.charAt(0));
        }
        if (recursive){
            dLog(caller,'return '+keyCodes.join(' '));
            return keyCodes;
        }
        if (keyCodes.length > 1 && keyCodes.length < 3) {
            dLog(caller,'full width, appending:'+keyCodes.join(''));
            appendHexStringArray(keyCodes);
        } else {
            dLog(caller,'half width, appending:'+keyCodes+"00");
            appendHexStringArray(keyCodes);
            appendHexStringArray(['00']);
        }
        return true;
    }

    function buildDelayByteTab(d) {
        let tmpTab = [];
        while (d > 0) {
            tmpTab.push('00');
            if (d > 255) {
                tmpTab.push('FF');
                d = d - 255
            } else {
                tmpTab.push(d.toString(16));
                d = 0
            }
        }

        /**  **/

        return tmpTab
    }

    /*  */
    function getBytesForKey(k,fullwidth=false) {
        let fw="";
        if (fullwidth){
            fw = " FULLWIDTH ";
        }
        dLog('LANGLOOKUP',fw+'get bytes for ' + k + " from "+IDE_CONFIG['ducky_lang_name']);
        let c = IDE_CONFIG['ducky_lang'][k];
        if (c === null || typeof c === 'undefined' || c === []) {
            dLog('LANGLOOKUP',c + ' NOT FOUND');
            return null;
        }
        let codes = c.split(',');
        // char
        let keyCodes;
        if (codes[2] !== '00') {
            keyCodes = [codes[2], codes[0]];
            dLog('LANGLOOKUP','KEY+MOD ' + keyCodes.join(" "));
            if (fullwidth){
                dLog('LANGLOOKUP','return ' + keyCodes.join(" "));
                return keyCodes;
            }
            if (keyCodes[1] === '00') {
                keyCodes.pop();
                dLog('LANGLOOKUP','KEY ONLY ' + keyCodes.join(' '));
            }
        } else {
            // modifier
            dLog('LANGLOOKUP','MODIFER - ' + codes[0] + ' of ' + codes.join(" "));
            // default take first byte
            keyCodes = [codes[0]];
        }
        dLog('LANGLOOKUP','default return ' + keyCodes.join(" "));
        return keyCodes;
    }

    /*  */
    function appendHexStringArray(hexStringArray) {
        if (hexStringArray === null) {
            console.error('cant append null');
            return;
        }
        let caller = 'appendHexStringArray()';
        let i = 0;
        let count = 0;
        for (i; i < hexStringArray.length; i++) {
            if (hexStringArray[i].length > 2) {
                oBB.push(hexStringArray[i].substring(0, 2));
                oBB.push(hexStringArray[i].substring(2, 4));
                count++;
                count++;
            } else {
                oBB.push(hexStringArray[i]);
                count++;
            }
        }
        let grouped = oBB.slice((oBB.length - count)).join('').match(/.{1,4}/g);
        let nospace = grouped.join('');
        let space = grouped.join(' ');
        dLog(caller, count + ' bytes added to inject.bin\n\n\t\t' + space + '\n\n');

        ///*  */

        if ((nospace.length % 4) !== 0) {
            if ((nospace.length % 2) !== 0) {
                let err = 'Encoder generated invalid number of bytes';
                throwError(err);
                return null;
            }
            dLog(caller,"Ensure encoder bytes valid");
        }
        return null;
    }

    function hexToByteArray(hexString) {
        let byteArray = new Uint8Array(hexString.length / 2);
        for (let x = 0; x < byteArray.length; x++) {
            byteArray[x] = parseInt(hexString.substr(x * 2, 2), 16);
        }
        return byteArray;
    }

        let str_regex = /^\s*STRING\s+.*$/;
        let l;
        if (inside_string_block || inside_stringln_block) {
            if (line.match(end_string_regex)) {
                inside_string_block = false;
                inside_stringln_block = false;
            } else {
                isInjectionString = true;
            }
        } else {
            if(line.match(str_regex)||line.match(strln_regex)||line.match(inline_string_regex)||line.match(inline_stringln_regex)) {
                isInjectionString = true;
            } else if(line.match(string_block_regex)||line.match(stringln_block_regex)) {
                inside_string_block = true;
                inside_stringln_block = true;
                isInjectionString = true;
            } else {
                line = trimInput(line);
            }
        }
        l = line.split('');

        for (let i = 0; i < l.length; i++) {
            let c = l[i];
            let c1 = l[i + 1]; // 1 char lookahead for doublechar operators
            if ((isOperator(c)) || ((c === '(') || (c === ')'))) {
                if (doubleOperator.includes((c + c1))) {
                    i++;//i++;
                } else {
                    if (!isInjectionString){
                        dLog('lex', "\tline[" + i + "]:\tfix " + c + ' spacing');
                        l[i] = ' ' + c + ' ';
                    }
                }
            }
        }
        if (isInjectionString){
            return line.split(' ');
        }
        l = l.join('');
        l = l.split(' ');
        return l.filter(Boolean);
    }

    function decToHex(i) {
        //dLog("decToHex (raw input) "+i);
        let result = "0000";
        i = parseInt(i, 10);
        if (i >= 0 && i <= 15) {
            result = "000" + i.toString(16);
        } else if (i >= 16 && i <= 255) {
            result = "00" + i.toString(16);
        } else if (i >= 256 && i <= 4095) {
            result = "0" + i.toString(16);
        } else if (i >= 4096 && i <= 65535) {
            result = i.toString(16);
        } else {
            console.error('FAILED decToHex (d) ' + i + '->' + result);
            return i;
        }
        //dLog('decToHex  (input)', i + '-> (output)' + result);
        return result;
    }

    async function sha256(message) {
        const msgBuffer = new TextEncoder().encode(message);
        try {
            const hashBuffer = await crypto.subtle.digest('SHA-256', msgBuffer);
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
        } catch {
           console.warn("SHA256SUM failed");
        }
        return "sha256sum-failed";

        function initPTour(element) {
            // define position functions
            setPositionFn(element);

            // collect all targets
            collectTargets(element);

            // set layout type
            updateLayout(element);

            // detect click on tour element
            element.element.addEventListener('click', function(event) {
                // check if we need to close tour
                var closeBtn = event.target.closest('.js-p-tour__close');
                if(closeBtn) closeTour(element);

                // check if this was a next/prev button
                var nextStep = event.target.closest('.js-p-tour__btn-next');
                if(nextStep) updateSteps(element, 'next');
                var prevStep = event.target.closest('.js-p-tour__btn-prev');
                if(prevStep) updateSteps(element, 'prev');
            });

            // toggle tour visibility
            element.showPTour ? showTour(element) : closeTour(element);

            // on resize -> update layout type
            window.addEventListener('resize', function(event){
                clearTimeout(element.resizingId);
                element.resizingId = setTimeout(function(){
                    updateLayout(element);
                    if(!element.mobile) showSelectedStep(element);
                }, 200);
            });

            // start tour using a trigger button
            if(element.tourTriggers.length > 0) {
                for(var i = 0; i < element.tourTriggers.length; i++) {
                    element.tourTriggers[i].addEventListener('click', function(event) {
                        updateLayout(element);
                        resetTour(element);
                        showTour(element);
                    });
                }
            }

            // add loaded class
            Util.addClass(element.element, 'p-tour--loaded');
        };

    var panels = [];
    var SlidingPanels = function(element) {
        this.element = element;
        this.itemsList = this.element.getElementsByClassName('js-s-panels__projects-list');
        this.items = this.itemsList[0].getElementsByClassName('js-s-panels__project-preview');
        this.navigationToggle = this.element.getElementsByClassName('js-s-panels__nav-control');
        this.navigation = this.element.getElementsByClassName('js-s-panels__nav-wrapper');
        this.transitionLayer = this.element.getElementsByClassName('js-s-panels__overlay-layer');
        this.selectedSection = false; // will be used to store the visible project content section
        this.animating = false;
        // aria labels for the navigationToggle button
        this.toggleAriaLabels = ['Toggle navigation', 'Close Project'];
        initSlidingPanels(this);
    };

    function initSlidingPanels(element) {
        // detect click on toggle menu
        if(element.navigationToggle.length > 0 && element.navigation.length > 0) {
            element.navigationToggle[0].addEventListener('click', function(event) {
                if(element.animating) return;
                // hideElement('devicemenu');
                // if project is open -> close project
                // if(closeProjectIfVisible(element)) return;

                // toggle navigation
                var openNav = Util.hasClass(element.navigation[0], 'is-hidden');
                toggleNavigation(element, openNav);
            });
        }

        // open project
        element.element.addEventListener('click', function(event) {
            if(element.animating) return;
            var link = event.target.closest('.js-s-panels__project-control');
            if(!link) return;
            event.preventDefault();
            openProject(element, event.target.closest('.js-s-panels__project-preview'), link.getAttribute('href').replace('#', ''));
        });
    };

    function closeProjectIfVisible(element) {

        if(visibleProject.length > 0) {
            element.animating = true;
            closeProject(element);
            return true;
        }

        return false;
    };

    function toggleNavigation(element, openNavigation,slide=true) {
        if (element)
            element.animating = true;
        if(openNavigation) Util.removeClass(element.navigation[0], 'is-hidden');
        !openNavigation ? Util.removeClass(document.getElementById('devicemenubutton'), 'hidden') : Util.addClass(document.getElementById('devicemenubutton'), 'hidden');
        slideProjects(element, openNavigation, false, function(){
            element.animating = false;
            if(!openNavigation) Util.addClass(element.navigation[0], 'is-hidden');
        },slide);
    };

    function openProject(element, project, id) {
        let mode = CHANGEMODE(id);
        var openNav = Util.hasClass(element.navigation[0], 'is-hidden');
        toggleNavigation(element, openNav);
        flashMessage('Editor Mode set to '+mode.label);

    function closeProject(element) {
        // remove transitions from projects
        toggleTransitionProjects(element, true);
        // hide navigation
        Util.removeClass(element.itemsList[0], 'bg-opacity-0');
        // reveal transition layer
        Util.addClass(element.transitionLayer[0], 's-panels__overlay-layer--visible');
        // wait for end of transition layer effect
        element.transitionLayer[0].addEventListener('transitionend', function cb(event) {
            if(event.propertyName != 'opacity') return;
            element.transitionLayer[0].removeEventListener('transitionend', cb);
            // update projects classes
            resetProjects(element);

            setTimeout(function(){
                // hide transition layer
                Util.removeClass(element.transitionLayer[0], 's-panels__overlay-layer--visible');
                // reveal projects
                slideProjects(element, false, false, function() {
                    Util.addClass(element.itemsList[0], 'bg-opacity-0');
                    element.animating = false;
                });
            }, 200);
        });

        // modify toggle button appearance
        Util.removeClass(element.navigationToggle[0], 's-panels__nav-control--close');
        // modify toggle button aria-label
        element.navigationToggle[0].setAttribute('aria-label', element.toggleAriaLabels[0]);
    };

    function slideProjects(element, openNav, exclude, cb,slide=true) {
        if (!slide){
            for(var i = 0; i < element.items.length; i++) {
                let e = element.items[i];
                if (!slide) {
                    e.classList.add('notransition'); // Disable transitions
                }
                Util.toggleClass(element.items[i], 's-panels__project-preview--hide', openNav);
                Util.toggleClass(document.getElementById('spanelcontainer'),'s-panels__project-preview--hide',openNav);
                e.offsetHeight; // Trigger a reflow, flushing the CSS changes
                e.classList.remove('notransition'); // Re-enable transitions
                toggleProjectAccessibility(element.items[i], openNav);
            }
            return;
        }
        // projects will slide out in a random order
        var randomList = getRandomList(element.items.length, exclude);
        for(var i = 0; i < randomList.length; i++) {(function(i){
            setTimeout(function(){
                Util.toggleClass(element.items[randomList[i]], 's-panels__project-preview--hide', openNav);
                toggleProjectAccessibility(element.items[randomList[i]], openNav);
                if(cb && i == randomList.length - 1) {
                    // last item to be animated -> execute callback function at the end of the animation
                    element.items[randomList[i]].addEventListener('transitionend', function cbt() {
                        if(event.propertyName != 'transform') return;
                        if(cb) cb();
                        element.items[randomList[i]].removeEventListener('transitionend', cbt);
                    });
                }
            }, i*100);
        })(i);}
        if(openNav){
            setTimeout(function(){
                Util.toggleClass(document.getElementById('spanelcontainer'),'s-panels__project-preview--hide',openNav);
            },1000);
        } else{
            Util.toggleClass(document.getElementById('spanelcontainer'),'s-panels__project-preview--hide',openNav);
        }
    };

    function toggleTransitionProjects(element, bool) {
        // remove transitions from project elements
        for(var i = 0; i < element.items.length; i++) {
            Util.toggleClass(element.items[i], 's-panels__project-preview--no-transition', bool);
        }
    };

    function resetProjects(element) {
        // reset projects classes -> remove selected/no-transition class + add hide class
        for(var i = 0; i < element.items.length; i++) {
            Util.removeClass(element.items[i], 's-panels__project-preview--selected s-panels__project-preview--no-transition');
            Util.addClass(element.items[i], 's-panels__project-preview--hide');
        }

        // hide project content
        if(element.selectedSection) Util.addClass(element.selectedSection, 'is-hidden');
        element.selectedSection = false;
    };

    function getRandomList(maxVal, exclude) {
        // get list of random integer from 0 to (maxVal - 1) excluding (exclude) if defined
        var uniqueRandoms = [];
        var randomArray = [];

        function makeUniqueRandom() {
            // refill the array if needed
            if (!uniqueRandoms.length) {
                for (var i = 0; i < maxVal; i++) {
                    if(exclude === false || i != exclude) uniqueRandoms.push(i);
                }
            }
            var index = Math.floor(Math.random() * uniqueRandoms.length);
            var val = uniqueRandoms[index];
            // now remove that value from the array
            uniqueRandoms.splice(index, 1);
            return val;
        }

        for(var j = 0; j < maxVal; j++) {
            randomArray.push(makeUniqueRandom());
        }

        return randomArray;
    };

        }
    };
